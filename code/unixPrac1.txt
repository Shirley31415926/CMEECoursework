# 1
wc -l ../data/fasta/407228326.fasta ../data/fasta/407228412.fasta ../data/fasta/e_coli.fasta
# Explanation: "wc -l" counts newline characters, which is '\n'.
#              If the last line doesn't have '\n' at the end,
#              it would be one less than the actual displayed line count.
#              The two numbered FASTA files have no newline character
#              at the end of their final lines, so their counts are
#              one lower than their displayed line counts.
#              The result is the number of newline characters in each file.


# 2
tail -n +2 ../data/fasta/e_coli.fasta
# Explanation: The header is excluded because everything starts from the second line.
#              "tail -n +2" means output everything starting from line 2,
#              rather than counting backwards from the bottom.
#              The result is the E. coli sequence lines without the header.


# 3
tail -n +2 ../data/fasta/e_coli.fasta | tr -d '\n' | wc -c
# Explanation: Exclude the header and remove line-ending characters
#              by deleting '\n' with tr.
#              "wc -c" counts the bytes in the cleaned sequence.
#              Since the DNA sequence symbols are single-byte characters,
#              the result is the total number of sequence positions,
#              including ambiguous nucleotide symbols such as N, R, Y, M and W.


# 4
tail -n +2 ../data/fasta/e_coli.fasta | tr -d '\n' | grep -o 'ATGC' | wc -l
# Explanation: Exclude the header and remove line-ending characters
#              so that wrapping cannot hide a motif occurrence.
#              Use "grep -o" to output every occurrence of the literal motif ATGC.
#              Then use "wc -l" to count the occurrences.
#              Ambiguous symbols are kept in the sequence to avoid
#              accidentally joining separated bases.
#              The result is the total number of occurrences of ATGC.


# 5
seq=$(tail -n +2 ../data/fasta/e_coli.fasta | tr -d '\n'); a=$(printf '%s' "$seq" | grep -o 'A' | wc -l); t=$(printf '%s' "$seq" | grep -o 'T' | wc -l); g=$(printf '%s' "$seq" | grep -o 'G' | wc -l); c=$(printf '%s' "$seq" | grep -o 'C' | wc -l); printf '%s / %s\n' "$((a + t))" "$((g + c))" | bc -l
# Explanation: Store the cleaned sequence in a variable called seq.
#              Count every occurrence of the four canonical DNA bases
#              (A, T, G and C) using grep -o and wc -l.
#              Use Bash arithmetic to calculate A+T and G+C.
#              Use printf to format the two counts as a division expression.
#              Finally, use "bc -l" to perform decimal division.
#              The result is the AT/GC ratio, calculated as (A+T)/(G+C).
#              Ambiguous nucleotide symbols are excluded from this calculation.
